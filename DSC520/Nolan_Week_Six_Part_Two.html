##  0.0000000 0.0018362
## sample estimates:
##  odds ratio 
## 0.000172163
  • null hypothesis is using social does not result in lower cognitive ability. alpha = .05
  • alternative hypothesis
  • p value < 2.2e-16
  • reject null hypothesis . data suggests using social media is associated with lower cognitive ability
  • one sided test - we only looked at if the cognitive ability was lower.

binom.test(95, 100, p=0.5)
## 
##  Exact binomial test
## 
## data:  95 and 100
## number of successes = 95, number of trials = 100, p-value < 2.2e-16
## alternative hypothesis: true probability of success is not equal to 0.5
## 95 percent confidence interval:
##  0.8871651 0.9835681
## sample estimates:
## probability of success 
##                   0.95
  • alternative hypothesis is the coin is NOT fair. P(heads) <> .5
  • p value of coin showing up 95/100 times = p-value < 2.2e-16
  • reject the null hypothesis, the coin is not fair with these results
  • this is a two sided test, because deviation high or low matters.

# p value input GOES HERE
pval <- .025
# create a Gaussian probability curve
zvals <- seq(-3, 3, length=1001)
zpdf <- dnorm(zvals)

# plot the probability function and the vertical lines
df1 <- data.frame(x=zvals, y=zpdf)
fig_A <- ggplot(df1, aes(x=x, y=y)) +
           geom_smooth(color="black") +
           scale_x_continuous(limit=c(head(zvals, 1), tail(zvals, 1)), expand=c(0, 0)) +
           scale_y_continuous(limit=c(0, .42), breaks=NULL, expand=c(0, 0)) +
           labs(x="", y=expression(paste("Prob. given ", "H"[0]))) +
           theme_classic()

# two-tailed p-values
mn_idx <- which.min(zvals**2)
pvalsL <- pnorm(zvals[1:mn_idx])
pvalsR <- 1 - pnorm(zvals[(mn_idx+1):length(zvals)])
pvals2 <- 2 * c(pvalsL, pvalsR)

# plot the probability function
df2 <- data.frame(x=zvals, y=pvals2)
fig_B <- ggplot(df2, aes(x=x, y=y)) +
           geom_line(color="black") +
           geom_hline(yintercept=pval, color="gray50", linetype="dashed") +
           scale_x_continuous(limit=c(head(zvals, 1), tail(zvals, 1)), expand=c(0, 0)) +
           labs(x="", y="P-value") +
           theme_classic()

# draw vertical lines
zcrit_low <- qnorm(pval/2)
zcrit_high <- qnorm(1 - pval/2)

vline1 <- geom_vline(aes(xintercept=zvals[which.min((zvals-zcrit_low)**2)]), color="darkgray",
                     linetype="dotted", linewidth=.75)
vline2 <- geom_vline(aes(xintercept=zvals[which.min((zvals-zcrit_high)**2)]), color="darkgray",
                     linetype="dotted", linewidth=.75)
fig_A <- fig_A + vline1 + vline2
fig_B <- fig_B + vline1 + vline2

# draw patches for significant regions
zidx <- which.min((zvals-zcrit_low)**2)
fig_A <- fig_A + geom_ribbon(data=subset(df1, x <= x[zidx]), aes(ymin=0, ymax=y), alpha=.4)
fig_B <- fig_B + geom_ribbon(data=subset(df2, x <= x[zidx]), aes(ymin=0, ymax=y), alpha=.4)

zidx <- which.min((zvals-zcrit_high)**2)
fig_A <- fig_A + geom_ribbon(data=subset(df1, x >= x[zidx]), aes(ymin=0, ymax=y), alpha=.4)
fig_B <- fig_B + geom_ribbon(data=subset(df2, x >= x[zidx]), aes(ymin=0, ymax=y), alpha=.4)

fig_C <- fig_B

# indicators
fig_A <- fig_A + annotate("text", x=0, y=.3, label='\"Not Significant\"') +
  annotate("segment", x=-2.35, xend=-2.2, y=.12, yend=.05, lwd=1.25, linejoin="mitre",
